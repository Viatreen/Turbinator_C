cmake_minimum_required(VERSION 3.20)
project(TurbinatorC VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(TURBINATOR_ENABLE_CUDA "Build optional CUDA C feature kernels when a CUDA compiler is available" ON)
option(TURBINATOR_P3000_MODE "Build the low-memory CUDA sensor + compact C11 learner configuration" OFF)
option(TURBINATOR_ENABLE_LARGE_GPU "Build the 618M CUDA learner" ON)
option(TURBINATOR_P3000_AVX2 "Use AVX2 instructions for the P3000 laptop CPU learner" ON)
option(TURBINATOR_BUILD_TESTS "Build native C tests" ON)
option(TURBINATOR_BUILD_TRAINER "Build native C training and monitoring CLI" ON)
option(TURBINATOR_BUILD_VIEWER "Build the native C desktop viewer on Windows" ON)
set(TURBINATOR_CUDA_ARCHITECTURES "70" CACHE STRING "CUDA GPU architectures (70 targets Tesla V100)")
if(TURBINATOR_P3000_MODE)
  if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  endif()
  set(TURBINATOR_CUDA_ARCHITECTURES "61" CACHE STRING "CUDA GPU architectures" FORCE)
  set(TURBINATOR_ENABLE_LARGE_GPU OFF CACHE BOOL "Build the 618M CUDA learner" FORCE)
  if(NOT TURBINATOR_ENABLE_CUDA)
    message(FATAL_ERROR "TURBINATOR_P3000_MODE requires TURBINATOR_ENABLE_CUDA=ON")
  endif()
  message(STATUS "P3000 configuration: sm_61, batched CUDA sensor features, compact C11 learner")
endif()

if(TURBINATOR_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    set(CMAKE_CUDA_STANDARD 11)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    set(TURBINATOR_HAS_CUDA ON)
  else()
    message(STATUS "CUDA compiler unavailable; building the CPU C implementation")
  endif()
endif()
if(TURBINATOR_P3000_MODE AND NOT TURBINATOR_HAS_CUDA)
  message(FATAL_ERROR "P3000 configuration requires a CUDA compiler")
endif()

add_library(turbinator_sim SHARED
  src/sim/physics.c
  src/sim/sensors.c
  src/sim/c_api.c
  src/trainer/tui.c
)
target_include_directories(turbinator_sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(turbinator_sim PRIVATE TURBINATOR_SIM_BUILD)

add_library(turbinator_training STATIC src/training/training.c)
target_include_directories(turbinator_training PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_library(turbinator_network STATIC src/network/network.c)
target_include_directories(turbinator_network PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(TURBINATOR_HAS_CUDA)
  add_library(turbinator_cuda STATIC src/cuda/features.cu)
  target_include_directories(turbinator_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  set_target_properties(turbinator_cuda PROPERTIES CUDA_ARCHITECTURES "${TURBINATOR_CUDA_ARCHITECTURES}")
  set_target_properties(turbinator_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
  target_compile_definitions(turbinator_cuda PUBLIC TURBINATOR_HAS_CUDA=1)

  if(TURBINATOR_ENABLE_LARGE_GPU)
    add_library(turbinator_gpu_training STATIC
    src/gpu/ops.cu
    src/gpu/optimizer.cu
    src/gpu/replay.cu
    src/gpu/sac_loss.cu
    src/gpu/trainer.cu
    src/network/network_gpu.cu
  )
    target_include_directories(turbinator_gpu_training PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_link_libraries(turbinator_gpu_training PUBLIC turbinator_network CUDA::cudart CUDA::cublas)
    target_compile_definitions(turbinator_gpu_training PUBLIC TURBINATOR_HAS_LARGE_GPU=1)
    set_target_properties(turbinator_gpu_training PROPERTIES
      CUDA_ARCHITECTURES "${TURBINATOR_CUDA_ARCHITECTURES}"
      CUDA_SEPARABLE_COMPILATION ON)
  endif()
endif()

if(MSVC)
  target_compile_options(turbinator_sim PRIVATE /W3)
  target_compile_options(turbinator_training PRIVATE /W3)
  target_compile_options(turbinator_network PRIVATE /W3)
else()
  target_compile_options(turbinator_sim PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(turbinator_training PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(turbinator_network PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(turbinator_sim PRIVATE m)
  target_link_libraries(turbinator_training PUBLIC m)
  target_link_libraries(turbinator_network PUBLIC m)
endif()

if(TURBINATOR_P3000_MODE AND TURBINATOR_P3000_AVX2 AND
   CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|x86_64|X86|i[3-6]86)$")
  if(MSVC)
    target_compile_options(turbinator_sim PRIVATE /arch:AVX2)
    target_compile_options(turbinator_training PRIVATE /arch:AVX2)
  else()
    target_compile_options(turbinator_sim PRIVATE -mavx2)
    target_compile_options(turbinator_training PRIVATE -mavx2)
  endif()
endif()

if(WIN32)
  target_link_libraries(turbinator_sim PRIVATE ws2_32)
endif()

if(ANDROID)
  add_library(turbinator_mobile SHARED src/android/native_activity.c)
  target_include_directories(turbinator_mobile PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(turbinator_mobile PRIVATE turbinator_sim android log m)
  target_link_options(turbinator_sim PRIVATE "-Wl,-z,max-page-size=16384")
  target_link_options(turbinator_mobile PRIVATE "-Wl,-z,max-page-size=16384")
endif()

if(TURBINATOR_BUILD_TRAINER)
  add_executable(turbinator_trainer src/trainer/main.c src/trainer/http.c)
  target_include_directories(turbinator_trainer PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(turbinator_trainer PRIVATE turbinator_sim turbinator_training turbinator_network)
  if(WIN32)
    target_link_libraries(turbinator_trainer PRIVATE ws2_32)
  endif()
  if(TURBINATOR_HAS_CUDA)
    target_link_libraries(turbinator_trainer PRIVATE turbinator_cuda)
  endif()
  if(TARGET turbinator_gpu_training)
    target_link_libraries(turbinator_trainer PRIVATE turbinator_gpu_training)
  endif()
  add_custom_command(TARGET turbinator_trainer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      $<TARGET_FILE:turbinator_trainer>
      $<TARGET_FILE_DIR:turbinator_trainer>/turbinator_tui$<TARGET_FILE_SUFFIX:turbinator_trainer>)
endif()

if(WIN32 AND TURBINATOR_BUILD_VIEWER)
  add_executable(turbinator_viewer WIN32 src/viewer/viewer_win32.c)
  target_link_libraries(turbinator_viewer PRIVATE user32 gdi32)
endif()

if(TURBINATOR_BUILD_TESTS)
  include(CTest)
  add_executable(turbinator_tests tests/test_native.c)
  target_link_libraries(turbinator_tests PRIVATE turbinator_sim turbinator_training)
  add_test(NAME native_smoke COMMAND turbinator_tests)
  add_executable(turbinator_training_tests tests/test_training.c)
  target_link_libraries(turbinator_training_tests PRIVATE turbinator_training)
  add_test(NAME training_sac COMMAND turbinator_training_tests)
  add_executable(turbinator_network_tests tests/test_network.c)
  target_link_libraries(turbinator_network_tests PRIVATE turbinator_network)
  add_test(NAME network_layout COMMAND turbinator_network_tests)
  if(TURBINATOR_HAS_CUDA)
    add_executable(turbinator_cuda_feature_tests tests/test_cuda_features.c)
    target_link_libraries(turbinator_cuda_feature_tests PRIVATE turbinator_cuda turbinator_training)
    add_test(NAME cuda_feature_batch COMMAND turbinator_cuda_feature_tests)
    add_executable(turbinator_onboard_features_tests tests/test_onboard_features.cu)
    target_include_directories(turbinator_onboard_features_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/network)
    target_link_libraries(turbinator_onboard_features_tests PRIVATE CUDA::cudart)
    set_target_properties(turbinator_onboard_features_tests PROPERTIES
      CUDA_ARCHITECTURES "${TURBINATOR_CUDA_ARCHITECTURES}")
    add_test(NAME onboard_features COMMAND turbinator_onboard_features_tests)
  endif()
  if(TARGET turbinator_gpu_training)
    add_executable(turbinator_gpu_tests tests/test_gpu.c)
    target_link_libraries(turbinator_gpu_tests PRIVATE turbinator_gpu_training)
    add_test(NAME gpu_ops COMMAND turbinator_gpu_tests)
    add_executable(turbinator_network_gpu_smoke tests/test_network_gpu.c)
    target_link_libraries(turbinator_network_gpu_smoke PRIVATE turbinator_gpu_training turbinator_network)
  endif()
endif()
